unsigned char *encode(unsigned char *src,int srclen)
{
int n,buflen,i,j;
int pading=0;
unsigned char *buf;
static unsigned char *dst; 

buf=src;
buflen=n=srclen;
if(n%3!=0)&nbsp;&nbsp;/* pad with '=' by using a temp buffer */
{
&nbsp;&nbsp;pading=1;
&nbsp;&nbsp;buflen=n+3-n%3;
&nbsp;&nbsp;buf=malloc(buflen+1);
&nbsp;&nbsp;memset(buf,0,buflen+1);
&nbsp;&nbsp;memcpy(buf,src,n);
&nbsp;&nbsp;for(i=0;i&lt;3-n%3;i++)
&nbsp; &nbsp;buf[n+i]='=';
}
dst=malloc(buflen*4/3+1);
memset(dst,0,buflen*4/3+1);
for(i=0,j=0;i&lt;buflen;i+=3,j+=4)
{
&nbsp;&nbsp;dst[j]=(buf&amp;0xFC)&gt;&gt;2;
&nbsp;&nbsp;dst[j+1]=((buf&amp;0x03)&lt;&lt;4) + ((buf[i+1]&amp;0xF0)&gt;&gt;4);
&nbsp;&nbsp;dst[j+2]=((buf[i+1]&amp;0x0F)&lt;&lt;2) + ((buf[i+2]&amp;0xC0)&gt;&gt;6);
&nbsp;&nbsp;dst[j+3]=buf[i+2]&amp;0x3F; 
}
for(i=0;i&lt;buflen*4/3;i++) /* map 6 bit value to base64 ASCII character */
&nbsp;&nbsp;dst=ch64[dst];
if(pading)
&nbsp;&nbsp;free(buf);
return dst;
}

unsigned char *decode(unsigned char *src)
{
int n,i,j;
unsigned char *p;
static unsigned char *dst; 

n=strlen(src);
for(i=0;i&lt;n;i++) /* map base64 ASCII character to 6 bit value */
{
&nbsp;&nbsp;p=strchr(ch64,src);
&nbsp;&nbsp;if(!p)
&nbsp; &nbsp;break;
&nbsp;&nbsp;src=p-ch64;
}
dst=malloc(n*3/4+1);
memset(dst,0,n*3/4+1);
for(i=0,j=0;i&lt;n;i+=4,j+=3)
{
&nbsp;&nbsp;dst[j]=(src&lt;&lt;2) + ((src[i+1]&amp;0x30)&gt;&gt;4);
&nbsp;&nbsp;dst[j+1]=((src[i+1]&amp;0x0F)&lt;&lt;4) + ((src[i+2]&amp;0x3C)&gt;&gt;2);
&nbsp;&nbsp;dst[j+2]=((src[i+2]&amp;0x03)&lt;&lt;6) + src[i+3];
}
return dst;
}
